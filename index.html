insert into auth.identities (id, user_id, provider_id, identity_data, provider, last_sign_in_at, created_at, updated_at) values (gen_random_uuid(), v_uid, v_uid::text, jsonb_build_object('sub', v_uid::text, 'email', v_mail, 'email_verified', true), 'email', now(), now(), now()); else update auth.users set email = v_mail, email_confirmed_at = coalesce(email_confirmed_at, now()), updated_at = now() where id = v_uid and lower(email) <> v_mail; end if; update members set user_id = v_uid where id = m.id and user_id is distinct from v_uid; delete from tw_login_tries where nid = p_nid; return jsonb_build_object('email', v_mail, 'masked', left(v_mail, 2) || repeat('•', greatest(position('@' in v_mail) - 3, 1)) || substr(v_mail, position('@' in v_mail))); create or replace function public.tw_guard_roles() returns trigger language plpgsql security definer set search_path = public as $$ begin if auth.uid() is null then return new; end if; if tg_op = 'UPDATE' and new.is_super is distinct from old.is_super then raise exception 'only_super'; end if; if tg_op = 'INSERT' and coalesce(new.is_super, false) then raise exception 'only_super'; end if; if (tg_op = 'INSERT' and coalesce(new.is_admin, false)) or (tg_op = 'UPDATE' and (new.is_admin is distinct from old.is_admin or new.admin_banks is distinct from old.admin_banks)) then if not exists (select 1 from members where user_id = auth.uid() and is_super) then raise exception 'only_super'; end if; if tg_op = 'UPDATE' and old.is_super then raise exception 'super_target'; end if; end if; return new; end $$; drop trigger if exists tw_guard_roles on public.members; create trigger tw_guard_roles before insert or update on public.members for each row execute function public.tw_guard_roles(); create or replace function public.tw_set_admin(p_member bigint, p_admin boolean) returns text language plpgsql security definer set search_path = public as $$ declare t record; begin if not exists (select 1 from members where user_id = auth.uid() and is_super) then return 'only_super'; end if; update members set is_admin = p_admin where id = p_member; return 'ok'; end $$; revoke all on function public.tw_set_admin(bigint, boolean) from public; grant execute on function public.tw_set_admin(bigint, boolean) to authenticated;
